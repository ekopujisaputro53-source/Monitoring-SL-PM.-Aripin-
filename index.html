// Ganti URL di bawah ini dengan link CSV yang Anda dapatkan dari Google Spreadsheet const GOOGLE_SHEET_CSV_URL = 'https://docs.google.com/spreadsheets/d/102xqbuIstUGkxxldOBCv161fjO16rnTRhOQUU-wFS7w/export?format=csv'; // Fungsi untuk mengambil dan memuat data dari Google Spreadsheet async function loadDataFromGoogleSheet() { try { const response = await fetch(GOOGLE_SHEET_CSV_URL); if (!response.ok) { throw new Error('Gagal terhubung ke Google Spreadsheet'); } const csvText = await response.text(); // Mengubah teks CSV menjadi format JSON / Array Object parseCSVToDataset(csvText); } catch (error) { console.error("Error memuat data spreadsheet:", error); alert("Gagal memuat data dari Google Spreadsheet. Menggunakan data lokal cadangan."); } } // Fungsi parser CSV sederhana untuk mencocokkan kolom dashboard Anda function parseCSVToDataset(csvText) { const lines = csvText.split('\n'); if (lines.length < 2) return; // Baris pertama adalah Header / Nama Kolom const headers = lines[0].split(',').map(h => h.trim().replace(/^"|"$/g, '')); let loadedData = []; for (let i = 1; i < lines.length; i++) { let line = lines[i].trim(); if (!line) continue; // Penanganan sederhana untuk data CSV yang mungkin mengandung koma di dalam tanda kutip let row = []; let inQuotes = false; let currentVal = ''; for (let char of line) { if (char === '"') { inQuotes = !inQuotes; } else if (char === ',' && !inQuotes) { row.push(currentVal.trim().replace(/^"|"$/g, '')); currentVal = ''; } else { currentVal += char; } } row.push(currentVal.trim().replace(/^"|"$/g, '')); // Mapping baris CSV ke struktur data dashboard Anda let item = { no: i, faId: row[1] || `FA-${1000 + i}`, nama: row[2] || '', alamat: row[3] || '', kelurahan: row[4] || '', noHp: row[5] || '', subkon: row[6] || '', jenis: row[7] || '', ab: row[8] || '', rencanaTgl: parseExcelDate(row[9] || ''), tglDikerjakan: parseExcelDate(row[10] || ''), status: (row[11] || 'PENDING').toUpperCase().trim(), keterangan: row[12] || '' }; if (item.faId && item.faId !== "FA ID") { loadedData.push(item); } } if (loadedData.length > 0) { mainDataset = loadedData; // Memperbarui data utama dashboard populateFilterOptions(); // Memperbarui opsi filter dropdown applyFilters(); // Merefresh tampilan tabel dan grafik console.log(`Berhasil memuat ${loadedData.length} data dari Google Spreadsheet!`); } }
